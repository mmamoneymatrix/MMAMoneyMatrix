"use client";

import { useState } from "react";
import { runScenario } from "@/lib/api";

export default function ScenarioPanel({ baseResult }) {
  const [scenarioResult, setScenarioResult] = useState(null);

  async function handleScenario(type) {
    const data = await runScenario(baseResult.fighterA, baseResult.fighterB, type);
    setScenarioResult(data);
  }

  return (
    <div className="border-t pt-4 space-y-4">
      <h3 className="text-xl font-semibold">Scenario Analysis</h3>

      <div className="flex gap-4 flex-wrap">
        <button
          onClick={() => handleScenario("short_notice")}
          className="bg-gray-700 text-white px-3 py-2 rounded"
        >
          Short Notice
        </button>

        <button
          onClick={() => handleScenario("altitude")}
          className="bg-gray-700 text-white px-3 py-2 rounded"
        >
          Altitude
        </button>

        <button
          onClick={() => handleScenario("five_round")}
          className="bg-gray-700 text-white px-3 py-2 rounded"
        >
          5-Round Fight
        </button>

        <button
          onClick={() => handleScenario("no_gym_bonus")}
          className="bg-gray-700 text-white px-3 py-2 rounded"
        >
          Remove Gym Bonus
        </button>
      </div>

      {scenarioResult && (
        <div className="mt-4 p-4 border rounded">
          <h4 className="font-semibold">Scenario Result</h4>
          <p>Fighter A Win Probability: {scenarioResult.winProbA}%</p>
          <p>Fighter B Win Probability: {scenarioResult.winProbB}%</p>
        </div>
      )}
    </div>
  );
}
