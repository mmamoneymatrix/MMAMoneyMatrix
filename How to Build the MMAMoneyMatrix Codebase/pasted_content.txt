3. Random fighter generator function
python
WEIGHT_CLASSES = [
    "flyweight", "bantamweight", "featherweight", "lightweight",
    "welterweight", "middleweight", "light_heavyweight", "heavyweight",
]

SPECIAL_TRAITS_POOL = [
    "iron_chin",
    "explosive_kicks",
    "elite_cardio",
    "slow_starter",
    "pressure_fighter",
    "counter_striker",
    "submission_hunter",
    "clinch_monster",
    "distance_sniper",
]

def rand_in_range(low, high):
    return round(random.uniform(low, high), 2)

def random_name():
    first = random.choice(
        ["Alex", "Rafael", "Jamal", "Li", "Carlos", "Mia", "Hana", "Jordan", "Tyrell", "Kei"]
    )
    last = random.choice(
        ["Silva", "Nguyen", "Johnson", "Santos", "Lee", "Garcia", "Khan", "Ito", "Brown", "Costa"]
    )
    nickname = random.choice(
        ["The Phantom", "The Reaper", "The Technician", "The Ghost", "The Sniper", "The Mauler"]
    )
    return f'{first} "{nickname}" {last}'

def random_height_reach(weight_class: str):
    base = {
        "flyweight": (160, 170, 160, 175),
        "bantamweight": (163, 173, 163, 178),
        "featherweight": (165, 178, 165, 183),
        "lightweight": (168, 180, 170, 188),
        "welterweight": (170, 185, 173, 193),
        "middleweight": (173, 188, 175, 198),
        "light_heavyweight": (178, 193, 180, 203),
        "heavyweight": (180, 205, 183, 210),
    }
    h_low, h_high, r_low, r_high = base[weight_class]
    return random.randint(h_low, h_high), random.randint(r_low, r_high)

def generate_random_fighter() -> Fighter:
    style = random.choice(STYLES)
    gender = random.choice(GENDERS)
    ethnicity = random.choice(ETHNICITIES)
    weight_class = random.choice(WEIGHT_CLASSES)

    height_cm, reach_cm = random_height_reach(weight_class)
    profile = STYLE_PROFILES[style]

    traits = random.sample(SPECIAL_TRAITS_POOL, k=random.randint(1, 3))

    return Fighter(
        name=random_name(),
        style=style,
        gender=gender,
        ethnicity=ethnicity,
        height_cm=height_cm,
        reach_cm=reach_cm,
        weight_class=weight_class,
        striking_accuracy=rand_in_range(*profile["striking_accuracy"]),
        striking_power=rand_in_range(*profile["striking_power"]),
        grappling_accuracy=rand_in_range(*profile["grappling_accuracy"]),
        grappling_control=rand_in_range(*profile["grappling_control"]),
        submission_threat=rand_in_range(*profile["submission_threat"]),
        defense=rand_in_range(*profile["defense"]),
        movement=rand_in_range(*profile["movement"]),
        fight_iq=rand_in_range(*profile["fight_iq"]),
        cardio=rand_in_range(*profile["cardio"]),
        durability=rand_in_range(*profile["durability"]),
        special_traits=traits,
    )
